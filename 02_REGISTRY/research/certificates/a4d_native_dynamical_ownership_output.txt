    lambda hDerived =>
  let pairing := fun p d => (e0 p) d;
  let A := { dP := dPof b, dD := dDof b, star0 := S0of b, star1 := S1of b };
  let A' :=
    { dP := dPof (symmetry b), dD := dDof (symmetry b), star0 := S0of (symmetry b), star1 := S1of (symmetry b) };
  And.casesOn hDerived fun left right =>
    And.casesOn right fun hdP right =>
      And.casesOn right fun hdD right =>
        And.casesOn right fun hS0 hS1 =>
          have hdata :=
            of_eq_true
              (Eq.trans
                (congrFun' (congrArg Eq (congr (congr (congr (congrArg FinitePrimalDualHodgeData.mk hdP) hdD) hS0) hS1))

D0.Geometry.mixedAction_transform.{u_1, u_2, u_3, u_4} {P0 : Type u_1} {P1 : Type u_2} {D3 : Type u_3} {D4 : Type u_4}
  [Fintype P0] [Fintype P1] [Fintype D3] [Fintype D4] [DecidableEq P0] [DecidableEq P1] [DecidableEq D3]
  [DecidableEq D4] (e0 : FiniteRealCarrier P0 ≃ₗ[ℝ] Module.Dual ℝ (FiniteRealCarrier D4))
  (e1 : FiniteRealCarrier P1 ≃ₗ[ℝ] Module.Dual ℝ (FiniteRealCarrier D3))
  (Q0 : FiniteRealCarrier P0 ≃ₗ[ℝ] FiniteRealCarrier P0) (Q1 : FiniteRealCarrier P1 ≃ₗ[ℝ] FiniteRealCarrier P1)
  (A : FinitePrimalDualHodgeData P0 P1 D3 D4) (psi chi lambda : FiniteRealCarrier P0) :
  have R4 := dualAction e0 Q0;
  have R3 := dualAction e1 Q1;
  have A' :=
    { dP := movedPrimalDifferential Q0 Q1 A.dP, dD := movedDualDifferential R3 R4 A.dD,
      star0 := movedStar R4 Q0 A.star0, star1 := movedStar R3 Q1 A.star1 };
  have pairing := fun p d => (e0 p) d;
  mixedPrimalDualAction pairing A' (Q0 psi) (Q0 chi) (Q0 lambda) = mixedPrimalDualAction pairing A psi chi lambda
