        pivot = next((i for i in range(j, len(rows)) if rows[i][j]), None)
        if pivot is None:
            return Z
        if pivot != j:

            result = -result

                t = rows[i][j] / value
                rows[i] = [x - t * y for x, y in zip(rows[i], rows[j])]

def exact_gate(n, name, wires):
    d = 2 ** n
    if name == "G":
        A = [[ONE]]
        for j in range(n):
            A = kron(A, G if j == wires[0] else eye(2))
        return A
    if name != "CX":
        raise ValueError("fixture permits only original positive G/CX")
    A = [[Z] * d for _ in range(d)]
    for col in range(d):
        flip = ((col >> (n - 1 - wires[0])) & 1) << (n - 1 - wires[1])
        A[col ^ flip][col] = ONE
    return A


def replay(n, word):
    U = eye(2 ** n)
    for name, wires in word:
        U = mul(exact_gate(n, name, wires), U)
    return U


def clean(P):
    return {e: c for e, c in P.items() if c}


def padd(P, Q):
    R = dict(P)
    for e, c in Q.items():
        R[e] = R.get(e, F(0)) + c
    return clean(R)


def pscale(s, P):
    return clean({e: F(s) * c for e, c in P.items()})


def pmul(P, Q):
    R = {}
    for (i, j), c in P.items():
        for (k, l), d in Q.items():
            e = (i + k, j + l)
            R[e] = R.get(e, F(0)) + c * d
    return clean(R)


def derivative_I(P):
    return clean({(i - 1, j): i * c for (i, j), c in P.items() if i})
