        row, column = GROUP.index(g), GROUP.index(start)
        observed[row, column] += w * mark**n
        duration[row, column] += n*w

    blocks = {}
    for word in WORDS:
        reg = zeros(8)
        for j, g in enumerate(GROUP):
            reg[GROUP.index(prepend(word, g)), j] = 1
        blocks[word] = mark * reg * diag(*(coefficient(word, g) for g in GROUP))
    D = sum(blocks.values(), zeros(8))
    check(any(A*B != B*A for A in blocks.values() for B in blocks.values()),
          "positive noncommuting block operators")

    @lru_cache(None)
    def transfer(seen):
        # F is the stopped action, V counts remaining paid block visits.
        F, V = zeros(8), I
        for word, A in blocks.items():
            if word in seen:
                F += A
            else:
                child, visits = transfer(seen | {word})
                F += child*A
                V += visits*A
        check(F == I+(D-I)*V, "ordered pressure formula for every visited set")
        return F, V

    F, V = transfer(frozenset())
    check(F == observed, "stopped transfer equals the unweighted sum of every ordered path")
    check(ROW*D == mark*scale*ROW, "absolute block weight is the row eigenvalue")
    if scale == mark == 1:
        check(ROW*F == ROW and ROW*V == ROW*duration,
              "normalized pressure saturates while clock counts the matching block")
    if mark*scale < 1 and not state_dependent:
        R = (I-D).inv()
        pair = sum((A*R*A for A in blocks.values()), zeros(8))*R
        # All positive paths ending in a marked earlier match, including those
        # that would already have stopped. Row augmentation equals the 1-norm.
        check(all(entry >= 0 for entry in pair-F), "marked pairs contain all first repeats")
        a = [mark*coefficient(word, 1) for word in WORDS]
        s, s2, s3 = sum(a), sum(v*v for v in a), sum(v**3 for v in a)
        check(ROW*pair == s2/(1-s)**2*ROW, "full ordered pair impulse")
        error = (ROW*(pair-F))[0]
        check(0 < error <= 2*s3/(1-s)**3 + 3*s2**2/(1-s)**4,
              "earlier triple and disjoint pair bound")
    if state_dependent:
        check(any(len(set(ROW*A)) > 1 for A in blocks.values()),
              "total intertwiner does not require separate scalar word effects")
    else:
        # Independent commutative calculation over unordered distinct prefixes.
        elementary = [1] + [0]*len(WORDS)
        for word in WORDS:
            a = mark*coefficient(word, 1)
            for j in reversed(range(1, len(elementary))):
                elementary[j] += a*elementary[j-1]
        expected = sum(factorial(j)*s for j, s in enumerate(elementary))
        check(ROW*V == expected*ROW, "elementary symmetric occupancy")
    return (ROW*F)[0], (ROW*duration)[0], tuple(F[:, 0]), leaves // len(GROUP)
